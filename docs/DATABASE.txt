// Lesson Content Comment
// A reply is a comment whose parent_id points to the comment being replied to in the same LessonContent.
// Reply chains remain relationally intact; the API groups every descendant under its root for a two-level UI.
Table comment {
  id integer PK
  lesson_content_id integer // NOT NULL; FK lesson_content.id
  user_id integer // NOT NULL; FK user.id
  parent_id integer // FK comment.id ON DELETE CASCADE; default: null
  content string // NOT NULL
  created_at datetime // NOT NULL; default: lambda
  updated_at datetime // NOT NULL; default: lambda
  deleted_at datetime // default: null; soft-delete preserves reply threads.
  // INDEX(lesson_content_id, created_at)
  // INDEX(parent_id)
  // INDEX(user_id)
  // Service validates that parent_id belongs to the same lesson_content_id.
  // Deleting a root comment hard-deletes its entire reply tree through ON DELETE CASCADE.
  // A non-root comment with replies is soft-deleted; its API content becomes a tombstone projection.
}

// Direct one-course payment transaction. Cart and Order are outside this MVP.
Table transaction {
  id integer PK
  student_id integer // NOT NULL; FK user.id
  course_id integer // NOT NULL; FK courses.id
  amount decimal // NOT NULL; default: 0; immutable USD price snapshot, scale: 2.
  status PaymentStatus // NOT NULL; default: PENDING
  transaction_code string unique // NOT NULL; UNIQUE
  payos_code string unique // UNIQUE when provided by the real or simulated PayOS callback.
  payos_link string
  idempotency_key string unique // NOT NULL; UNIQUE per checkout command.
  signature_verified boolean // NOT NULL; default: False
  expires_at datetime
  completed_at datetime
  created_at datetime // NOT NULL; default: lambda
  updated_at datetime // NOT NULL; default: lambda
  // A transactionally locked service workflow expires/reuses stale rows before creating a new one.
  // At most one PENDING transaction with expires_at in the future may exist per (student_id, course_id).
  // A time-dependent rule cannot be represented by a static database UNIQUE constraint.
}
